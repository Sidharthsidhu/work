INNER JOIN

Only customers with orders

SELECT * FROM Customers c
INNER JOIN Orders o
  ON o.CustomerId = c.Id

LEFT JOIN

All customers, orders if any

SELECT * FROM Customers c
LEFT JOIN Orders o
  ON o.CustomerId = c.Id

RIGHT JOIN

All orders, customer if any

SELECT * FROM Customers c
RIGHT JOIN Orders o
  ON o.CustomerId = c.Id

FULL OUTER JOIN

Everything from both sides

SELECT * FROM Customers c
FULL OUTER JOIN Orders o
  ON o.CustomerId = c.Id
Tip: customers with no orders? LEFT JOIN + WHERE o.Id IS NULL
